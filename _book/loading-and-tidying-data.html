2.1 Import Sample metadata

mssm_sample_metadata <- airtable("4-MSE-Info", "appKakM1bnKSekwuW") %>%  get base ID from Airtable browser URL
  read_airtable(., fields = c(
    "ID", "LabBatch_text", "IntestinalSection", "SampleType",
    "Xcoord", "Ycoord", "SizeApprox", "cryosection_text", "buffer_text", "Collection_method_text",
    "Collection_Success", "Collection_attempts", "UsedCycles", "animal_temp", "Protocol_text",
    "adapter_type_text"
  ), id_to_col = TRUE) %>%
  filter(
    LabBatch_text %in% c(
       "MSEB0001", "MSEB0002", "MSEB0003", "MSEB0004", "MSEB0005",
      "MSEB0006", "MSEB0009", "MSEB0010", "MSEB0011", "MSEB0012",
      "MSEB0014", "MSEB0015"
    ),
    !SampleType %in% c("ControlHuman")
  ) %>%
  rename(
    batch = LabBatch_text, microsample = ID, section = IntestinalSection, 
    type = SampleType, cryosection = cryosection_text, 
    buffer = buffer_text, collection_method = Collection_method_text,
    collection = Collection_Success, collection_attempts = Collection_attempts,
    cycles = UsedCycles, animal = animal_temp, size = SizeApprox, 
    protocol = Protocol_text, adapter = adapter_type_text
  ) %>%
  select(
    microsample, section, type, batch, cryosection, buffer, Xcoord, Ycoord, 
    size, collection_method, collection, collection_attempts, cycles, animal, 
    protocol, adapter
  ) %>%
  mutate(
    type_binomial = ifelse(type == "Positive", "positive", "negative"),
    buffer = case_when(
      buffer == "B01" ~ "L01",
      buffer == "B02" ~ "L02",
      buffer == "B03" ~ "L03",
      buffer == "B04" ~ "L04",
      buffer == "B09" ~ "L05",
      buffer == "B11" ~ "L06",
      buffer == "B12" ~ "L07",
      buffer == "B13" ~ "L08",
      TRUE ~ buffer
    )
  ) %>%
  unnest(c(section, Xcoord, Ycoord, size, collection, cycles)) %>%
  arrange(microsample)

2.2 Sequencing Data Quality

2.2.1 Read Quality: Pre- vs Post-Filtering (fastp)

Note: The parse_fastp() function in index needs to be loaded.

mssm_json_data <- fromJSON("data/multiqc_data.json") # Download file from ERDA
mssm_fastp <- parse_fastp(mssm_json_data) %>%
  rename_with(~ gsub("^(summary|filtering_result|adapter_cutting|polyx_trimming)\\.", "", .)) %>%
  select(
    sample_id,
    before_filtering.total_reads,
    after_filtering.total_reads,
    before_filtering.total_bases,
    after_filtering.total_bases,
    before_filtering.gc_content,
    after_filtering.gc_content,
    low_quality_reads,
    adapter_trimmed_reads,
    adapter_trimmed_bases,
    duplication.rate
  ) %>%
  mutate(microsample = str_remove(sample_id, "\\.lib\\d+$")) %>%
  filter(grepl("^M", microsample)) %>%
  group_by(microsample) %>%
  summarise(
    across(contains("reads"), sum, .names = "total_{.col}"),
    across(contains("bases"), sum, .names = "total_{.col}"),
    across(ends_with("content"), mean, .names = "percent_{.col}"),
    across(ends_with("rate"), mean, .names = "percent_{.col}")
  )

2.2.2 Contamination Assessment: Host, Human, and Environmental Reads

mssm_euk_mapping <- read_tsv("https://sid.erda.dk/share_redirect/aTWbNfAf1o/results/preprocess_bowtie2_multiqc_report_data/multiqc_samtools_stats.txt") %>%
  mutate(
    reference = case_when(
      grepl("GRCh38", Sample, ignore.case = TRUE) ~ "human",
      grepl("GRCg7b", Sample, ignore.case = TRUE) ~ "chicken",
      grepl("Sscrofa11.1", Sample, ignore.case = TRUE) ~ "pig",
      TRUE ~ NA_character_
    ),
    microsample = str_extract(Sample, "M\\d+")
  ) %>%
  filter(
    !is.na(reference), # Remove rows where reference is NA
    grepl("^M", microsample)
  ) %>%
  select(Sample, microsample, reference, raw_total_sequences, reads_mapped, reads_unmapped) %>%
  group_by(microsample, reference) %>%
  summarise(
    raw_total_sequences = sum(raw_total_sequences, na.rm = TRUE),
    reads_mapped = sum(reads_mapped, na.rm = TRUE),
    reads_unmapped = sum(reads_unmapped, na.rm = TRUE)
  ) %>%
  pivot_wider(names_from = reference, values_from = c(raw_total_sequences, reads_mapped, reads_unmapped))

# download.file("https://sid.erda.dk/share_redirect/aTWbNfAf1o/results/quantify.zip", "data/quantify.zip", mode = "wb")
mssm_mag_mapping <- read.table(unz("data/quantify.zip", "multiqc_samtools_stats.txt"), header = T, sep = "\t", quote = "\"") %>%

  select(Sample, microsample, raw_total_sequences, reads_mapped, reads_unmapped) %>%
  group_by(microsample) %>%
  summarise(
    raw_total_sequences_mag = sum(raw_total_sequences, na.rm = TRUE),
    reads_mapped_mag = sum(reads_mapped, na.rm = TRUE),
    reads_unmapped_mag = sum(reads_unmapped, na.rm = TRUE)
  )

2.2.4 Read-Centric Mapping Statistics: Kraken (genomes) and SingleM (single-copy marker genes)

mssm_singlem <- read_tsv("https://sid.erda.dk/share_redirect/aTWbNfAf1o/results/preprocess/singlem/microbial_fraction.tsv") %>%
  mutate(microsample = str_extract(sample, "M\\d+")) %>%
  filter(grepl("^M", microsample)) %>%
  select(-c(warning, sample)) %>%
  group_by(microsample) %>%
  summarise(
    read_fraction_singlem = (mean(read_fraction, na.rm = TRUE)) / 100 #,
    # bacterial_archaeal_bases_singlem = sum(bacterial_archaeal_bases, na.rm = TRUE),
    # average_bacterial_archaeal_genome_size_singlem = mean(average_bacterial_archaeal_genome_size, na.rm = TRUE)
  )
mssm_kraken <- read_tsv("https://sid.erda.dk/share_redirect/aTWbNfAf1o/results/preprocess_kraken2_multiqc_report_data/multiqc_kraken.txt") %>%
  mutate(microsample = str_extract(Sample, "M\\d+")) %>%
  filter(grepl("^M", microsample)) %>%
  select(Sample, microsample, D) %>%
  separate(D, into = c("Eukaryota", "Bacteria", "Archaea", "Viruses"), sep = ", ", remove = TRUE) %>%
  mutate(across(!all_of("microsample"), ~ parse_number(.))) %>%
  group_by(microsample) %>%
  summarise(
    kr_Eukaryota = sum(Eukaryota, na.rm = TRUE),
    kr_Bacteria = sum(Bacteria, na.rm = TRUE),
    kr_Archaea = sum(Archaea, na.rm = TRUE),
    kr_Viruses = sum(Viruses, na.rm = TRUE),
  ) %>%
  mutate(
    total_kraken = kr_Eukaryota + kr_Bacteria + kr_Archaea + kr_Viruses,
    bacteria_pct_kraken = kr_Bacteria / total_kraken,
    eukaryota_pct_kraken = kr_Eukaryota / total_kraken
  )

2.2.5 Redundancy & Coverage Estimation: Nonpareil

mssm_nonpareil <- read_tsv("https://sid.erda.dk/share_redirect/aTWbNfAf1o/results/preprocess_nonpareil_multiqc_report_data/nonpareil-table.txt") %>%
  mutate(microsample = str_extract(Sample, "M\\d+")) %>%
  filter(grepl("^M", microsample)) %>%
  select(microsample, Redundancy, Coverage, Diversity) %>%
  group_by(microsample) %>%
  summarise(
    redundancy_nonpareil = (mean(Redundancy, na.rm = TRUE)) / 100,
    coverage_nonpareil = (mean(Coverage, na.rm = TRUE)) / 100,
    diversity_nonpareil = mean(Diversity, na.rm = TRUE)
  )

2.2.6 Combined Quality Statistics

mssm_quality <- reduce(
  list(
    mssm_sample_metadata,
    mssm_fastp,
    mssm_euk_mapping,
    mssm_mag_mapping,
    mssm_singlem,
    mssm_kraken,
    mssm_nonpareil
  ),
  full_join,
  by = "microsample"
) %>%
  mutate(
    lowquality_reads_pct = ((total_before_filtering.total_reads - total_after_filtering.total_reads) / total_before_filtering.total_reads),
    adapter_contamination_pct = (total_adapter_trimmed_bases / total_before_filtering.total_bases),
    host_pct = (reads_mapped_chicken / total_before_filtering.total_reads),
    human_pct = (reads_mapped_human / total_before_filtering.total_reads),
    pig_pct = (reads_mapped_pig / total_before_filtering.total_reads),
    prokreads_pct = (raw_total_sequences_mag / total_before_filtering.total_reads),
    mapmag_rawreads_pct = (reads_mapped_mag / total_before_filtering.total_reads),
    mapmag_filteredreads_pct = (reads_mapped_mag / total_after_filtering.total_reads),
    mapmag_prokreads_pct = (reads_mapped_mag / raw_total_sequences_mag)
  ) %>%
  filter(microsample %in% c(mssm_sample_metadata$microsample))
