  select(

  mutate(
    reference = case_when(
      grepl("GRCh38", Sample, ignore.case = TRUE) ~ "human",
      grepl("GRCg7b", Sample, ignore.case = TRUE) ~ "chicken",

  group_by(microsample, reference) %>%
  summarise(
    raw_total_sequences = sum(raw_total_sequences, na.rm = TRUE),
    reads_mapped = sum(reads_mapped, na.rm = TRUE),
    reads_unmapped = sum(reads_unmapped, na.rm = TRUE)

  pivot_wider(names_from = reference, values_from = c(raw_total_sequences, reads_mapped, reads_unmapped))

    contamination = Contamination,
    coding_density = Coding_Density,
    length = Genome_Size,
    N50_length = Contig_N50
  ) %>%
  # estimate genome circularity (N50 should be equal to genome length)
  mutate(circularity = N50_length / length) %>%
  mutate(circularity = ifelse(circularity >= 1, "1", "0")) %>%
  # update completeness and contamination for genomes with circularity of 1
  mutate(
    completeness = ifelse(circularity == "1", 100, completeness),
    contamination = ifelse(circularity == "1", 0, contamination)
  ) %>%
  # estimate ratio of: genome length / median bacterial genome length
  mutate(length_ratio = length / median(length, na.rm = TRUE))

2.3.1.3 Coverage Filtering & Genome Count Normalization

2.3.1.3.1 Coverage filtering

Note: Genomes that have less than 30% of their length covered by reads are turned into zeros to account for the random allocation of reads across genomes due to mapping heuristics.

macroquant_coverage <- macroquant_covered_bases %>% # calculate coverage values by dividing the number of covered bases by the length of each genome
  mutate(across(where(is.numeric), ~ . / macromag_genomemetadata$length))

macroquant_read_counts_filtered <- list()
macroquant_summary_filtered <- list()
# Define thresholds
coverage_thresholds <- c(0.4, 0.35, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05)
# Loop over thresholds
for (threshold in coverage_thresholds) {
  df <- macroquant_coverage %>%
    mutate(across(where(is.numeric), ~ ifelse(. > threshold, 1, 0))) %>%
    mutate(across(-1, ~ . * macroquant_read_counts[[cur_column()]]))
  summary <- df %>%
    summarise(across(where(is.numeric), \(x) sum(x, na.rm = TRUE))) %>%
    pivot_longer(
      cols = where(is.numeric), # columns to pivot (all numeric here)
      names_to = "microsample", # new column for former column names
      values_to = paste0("read_count_", threshold * 100) # new column for values
    )
  name <- paste0("filt_", threshold * 100)
  macroquant_read_counts_filtered[[name]] <- df
  macroquant_summary_filtered[[name]] <- summary
}
macroquant_merged_summary <- reduce(macroquant_summary_filtered, left_join, by = "microsample")

min_coverage <- 0.3
macroquant_read_counts_filt_30 <- macroquant_coverage %>%
  mutate(across(where(is.numeric), ~ ifelse(. > min_coverage, 1, 0))) %>%
  mutate(across(-1, ~ . * macroquant_read_counts[[cur_column()]]))

         N50_length = Contig_N50) %>%
  # estimate genome circularity (N50 should be equal to genome length)
  mutate(circularity = N50_length/length) %>%
  mutate(circularity = ifelse(circularity >= 1, "1", "0")) %>%
  # update completeness and contamination for genomes with circularity of 1
  mutate(completeness = ifelse(circularity == "1", 100, completeness),
         contamination = ifelse(circularity == "1", 0, contamination)) %>%   
  # estimate ratio of: genome length / median bacterial genome length
  mutate(length_ratio = length/median(length, na.rm = TRUE))

ani_files <- list.files("data/", pattern = "_population_ani.tsv$", full.names = TRUE)
# Return a named list of cleaned ANI matrices
ani_list <- purrr::map(ani_files, function(file_path) {
  # Extract bin ID
  bin_id <- stringr::str_extract(basename(file_path), "GPB_bin_\\d+")

  # Extract sample labels
  labels <- tibble(raw = readLines(file_path)) %>%
    filter(str_starts(raw, "##sample")) %>%
    mutate(
      ID = str_extract(raw, "(?<=ID=)\\d+") %>% as.integer(),
      SAMPLE = str_extract(raw, "M30[0-9]{4}")
    ) %>%
    select(ID, SAMPLE)

  # Read and clean ANI matrix
  ani <- readr::read_tsv(file_path, comment = "#", show_col_types = FALSE) %>%
    left_join(labels, by = c("SampleID" = "ID")) %>%
    select(-SampleID) %>%
    column_to_rownames(var = "SAMPLE") %>%
    set_names(labels$SAMPLE) %>%
    select(where(~ !all(is.na(.)))) %>%
    select(where(~ !all(is.nan(.)))) %>% 
    filter(!if_all(everything(), is.nan))

  return(ani)
})
